# Recode-IT backend Phase 3 dependencies.
#
# Stdlib-first design: SQLite via stdlib, no SQLAlchemy. Crypto and HTTP are
# the only non-stdlib deps. Email delivery (resend SDK) is loaded lazily so
# it's optional for tests / local dev.

starlette>=0.36,<1.0
httpx>=0.27          # required by starlette.testclient
PyJWT>=2.8,<3.0
argon2-cffi>=23.1
bcrypt>=4.1

# Dev / test
pytest>=8.0
pytest-asyncio>=0.23

# Email backend used in production (Resend). Imported lazily — when the
# module is absent OR `RESEND_API_KEY` env var is unset, the mailer falls
# back to stderr-printing the message body (dev / test mode). With both
# present, mails are sent live.
#
# Deploy doctrine: production systemd unit must `EnvironmentFile=/etc/mhc-server.env`
# to load `RESEND_API_KEY` (shared with MHC-L). Without that, mails are
# silently logged instead of sent — the failure is invisible to the client.
resend>=0.7

# Stripe SDK — used by stripe_webhook.py for signature verification via
# stripe.Webhook.construct_event. Imported lazily; tests bypass via
# RECODE_IT_WEBHOOK_TEST_BYPASS=1 (no SDK needed). In production this MUST
# be installed and STRIPE_WEBHOOK_SECRET set.
stripe>=10.0
